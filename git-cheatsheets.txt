
git config --global user.name "Имя" - задать имя пользователя.
git config --global user.email "email" - задать электронную почту.


git init -b main - инициализировать локальный репозиторий с веткой main.
git status - проверить текущее состояние файлов.


git add . - добавить все изменения в индекс.
git commit -m "сообщение" - создать коммит с описанием.


git log - посмотреть историю коммитов.
git log --oneline - компактный просмотр истории.
git show [хеш] - детальный просмотр изменений конкретного коммита.


git branch - список локальных веток.
git switch -c [имя_ветки] - создать ветку и перейти на неё.
git switch [имя_ветки] - переключиться на существующую ветку.
git branch -d [имя_ветки] - удалить слитую ветку.


git remote add origin [url] - связать локальный репозиторий с удаленным.
git remote -v - проверить адреса удаленного репозитория.
git push -u origin main - отправить изменения и связать ветки.
git pull - забрать актуальные изменения с сервера.
git fetch --prune - обновить список веток и удалить устаревшие ссылки.
